using App_Administration_RND.Data;
using App_Administration_RND.DTOs;
using App_Administration_RND.Models;
using App_Administration_RND.Services;
using Microsoft.AspNetCore.Mvc;
using Microsoft.EntityFrameworkCore;

namespace App_Administration_RND.Controllers
{
    [ApiController]
    [Route("api/[controller]")]
    public class AuthController : ControllerBase
    {
        private readonly ApplicationDbContext _context;
        private readonly PasswordService _passwordService;

        public AuthController(ApplicationDbContext context)
        {
            _context = context;
            _passwordService = new PasswordService();
        }

        // ✅ REGISTER
        [HttpPost("register")]
        public async Task<IActionResult> Register([FromBody] AuthRequest register)
        {
            if (string.IsNullOrWhiteSpace(register.Username) || string.IsNullOrWhiteSpace(register.Password))
                return BadRequest("Username dan Password wajib diisi.");

            // Cek apakah username sudah digunakan
            var existingUser = await _context.STGM_ADM_Users.FirstOrDefaultAsync(u => u.Username == register.Username);
            if (existingUser != null)
                return Conflict("Username sudah digunakan.");

            // Hash password
            string hashedPassword = _passwordService.HashPassword(register.Password);

            // Buat user baru
            var newUser = new ApplicationUser
            {
                Name = register.Username, // bisa disesuaikan, misal pakai input tambahan kalau ada
                Username = register.Username,
                PasswordHash = hashedPassword,
                LevelId = 7,
                AndroidAccess = 1,
                AndroidIsLocked = 0,
                ManId = "RAP",
                Email = $"{register.Username}@example.com",
                EtlStatus = null
            };

            _context.STGM_ADM_Users.Add(newUser);
            await _context.SaveChangesAsync();

            return Ok(new
            {
                success = true,
                message = "Registrasi berhasil.",
                username = newUser.Username
            });
        }

        // ✅ LOGIN
        [HttpPost("login")]
        public async Task<IActionResult> Login([FromBody] AuthRequest login)
        {
            if (string.IsNullOrWhiteSpace(login.Username) || string.IsNullOrWhiteSpace(login.Password))
                return BadRequest("Username dan Password wajib diisi.");

            var user = await _context.STGM_ADM_Users.FirstOrDefaultAsync(u => u.Username == login.Username);
            if (user == null)
                return Unauthorized("Username tidak ditemukan.");

            bool isValid = _passwordService.VerifyPassword(user.PasswordHash, login.Password);

            if (!isValid)
                return Unauthorized("Password salah.");

            // Simpan session
            HttpContext.Session.SetString("Username", user.Username);

            var response = new AuthResponse
            {
                Success = true,
                Message = "Login berhasil.",
                Username = user.Username,
                Name = user.Name,
                LevelId = user.LevelId
            };

            return Ok(response);
        }

        // ✅ LOGOUT
        [HttpPost("logout")]
        public IActionResult Logout()
        {
            HttpContext.Session.Clear();
            return Ok(new { message = "Logout berhasil." });
        }

        // ✅ CEK SESSION
        [HttpGet("session")]
        public IActionResult GetSession()
        {
            var username = HttpContext.Session.GetString("Username");
            if (username == null)
                return Unauthorized(new { message = "Belum login." });

            return Ok(new { message = "Session aktif", username });
        }
    }
}

        public async Task<ActionResult<object>> SyncDailyCheckins([FromBody] List<DailyCheckin> checkins)
        {
            var username = HttpContext.Session.GetString("Username");
            if (string.IsNullOrEmpty(username))
                return Unauthorized(new { message = "Session tidak ditemukan. Silakan login dulu." });

            if (checkins == null || !checkins.Any())
                return BadRequest(new { message = "Data sync tidak boleh kosong." });

            int uploaded = 0;

            foreach (var header in checkins)
            {
                var existing = await _context.STGM_ADM_DailyCheckins
                    .FirstOrDefaultAsync(x =>
                        x.Trial_Code == header.Trial_Code &&
                        x.CreateDate == header.CreateDate &&
                        x.CreateBy == username);

                if (existing == null)
                {
                    header.Id = 0;
                    header.CreateBy = username;

                    if (header.Details != null)
                    {
                        foreach (var detail in header.Details)
                        {
                            detail.Id = 0;
                        }
                    }

                    _context.STGM_ADM_DailyCheckins.Add(header);
                    uploaded++;
                }
            }

            await _context.SaveChangesAsync();

            return Ok(new
            {
                message = $"Berhasil sync {uploaded} data DailyCheckin ke server.",
                uploadedCount = uploaded
            });
        }
    }

using App_Administration_RND.Data;
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);

// ========== KONFIGURASI SERVER URLS ==========
// PENTING: Server akan listen di semua network interfaces
builder.WebHost.ConfigureKestrel(serverOptions =>
{
    // Listen pada semua IP addresses di port 7021
    serverOptions.ListenAnyIP(7021, listenOptions =>
    {
        listenOptions.UseHttps(); // Gunakan HTTPS
    });

    // Optional: Listen HTTP juga di port 7020 untuk fallback
    serverOptions.ListenAnyIP(7020);
});

// Atau gunakan cara sederhana (pilih salah satu):
// builder.WebHost.UseUrls("https://0.0.0.0:7021", "http://0.0.0.0:7020");

// ========== DATABASE CONTEXT ==========
builder.Services.AddDbContext<ApplicationDbContext>(options =>
    options.UseSqlServer(builder.Configuration.GetConnectionString("DefaultConnection")));

// ========== SESSION ==========
builder.Services.AddDistributedMemoryCache();
builder.Services.AddSession(options =>
{
    options.IdleTimeout = TimeSpan.FromHours(2);
    options.Cookie.HttpOnly = true;
    options.Cookie.IsEssential = true;
});

// ========== CORS (PENTING untuk Mobile App) ==========
builder.Services.AddCors(options =>
{
    options.AddPolicy("AllowAll", builder =>
        builder.AllowAnyOrigin()
               .AllowAnyHeader()
               .AllowAnyMethod());
});

builder.Services.AddControllers();
builder.Services.AddEndpointsApiExplorer();
builder.Services.AddSwaggerGen();

var app = builder.Build();

// ========== MIDDLEWARE ==========
if (app.Environment.IsDevelopment())
{
    app.UseSwagger();
    app.UseSwaggerUI();
}

// CORS harus di awal
app.UseCors("AllowAll");

// Untuk development, kita bisa skip HTTPS redirect jika perlu
// app.UseHttpsRedirection();

app.UseStaticFiles();

// Session harus sebelum Authorization
app.UseSession();

app.UseAuthorization();

// Redirect ke login.html jika akses root
app.MapGet("/", context =>
{
    context.Response.Redirect("/login.html");
    return Task.CompletedTask;
});

app.MapControllers();

// Log URL yang digunakan
var urls = app.Urls;
Console.WriteLine("========================================");
Console.WriteLine("Server berjalan di:");
foreach (var url in urls)
{
    Console.WriteLine($"  - {url}");
}
