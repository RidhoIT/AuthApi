        // 🔹 Ambil data Check-In
        async function loadCheckInData() {
            const container = document.getElementById("checkinContainer");
            container.innerHTML = "<p class='loading'>Loading data...</p>";

            try {
                const res = await fetch("/api/DailyCheckInApi");
                if (!res.ok) throw new Error("Gagal mengambil data check-in");

                const json = await res.json();
                const data = json.data;
                container.innerHTML = "";

                if (!data || data.length === 0) {
                    container.innerHTML = "<p class='empty'>Tidak ada data check-in.</p>";
                    return;
                }

                data.forEach((item, index) => {
                    const card = document.createElement("div");
                    card.className = "checkin-card";
                    card.innerHTML = `
                            <h3>Check-In #${index + 1}</h3>
                            <div class="checkin-info"><strong>Date:</strong> ${new Date(item.createDate).toLocaleString()}</div>
                            <div class="checkin-info"><strong>Location:</strong> ${item.location || "-"}</div>
                            <div class="checkin-info"><strong>Trial Code:</strong> ${item.trial_Code || "-"}</div>
                            <div class="checkin-info"><strong>Status:</strong> ${item.status || "-"}</div>

                            <button class="delete-btn" style="background-color:red;">Delete</button>
                        `;

                    // Lihat detail
                    card.querySelector(".view-btn").onclick = () => {
                        window.location.href = `checkin_details.html?id=${item.id}`;
                    };

                    // Hapus check-in
                    card.querySelector(".delete-btn").onclick = async (e) => {
                        e.stopPropagation();
                        if (!confirm("Yakin ingin menghapus check-in ini beserta semua details?")) return;

                        try {
                            const delRes = await fetch(`/api/DailyCheckInApi/${item.id}`, {
                                method: "DELETE"
                            });
                            const delJson = await delRes.json();
                            if (!delRes.ok) throw new Error(delJson.message || "Gagal menghapus data");

                            loadCheckInData();
                        } catch (err) {
                            console.error(err);
                            alert("Gagal menghapus data check-in");
                        }
                    };

                    container.appendChild(card);
                });
            } catch (err) {
                console.error("Error load data:", err);

                }

        }

    <script>
        document.getElementById("loginForm").addEventListener("submit", async function (e) {
            e.preventDefault();
            const msg = document.getElementById("msg");
            msg.style.color = "red";
            msg.textContent = "Memproses login...";

            try {
                const response = await fetch("/api/auth/login", {
                    method: "POST",
                    headers: {
                        "Content-Type": "application/json"
                    },
                    credentials: "include", // penting untuk ASP.NET session
                    body: JSON.stringify({
                        username: document.getElementById("Username").value.trim(),
                        password: document.getElementById("Password").value.trim()
                    })
                });

                const text = await response.text(); // ambil response mentah
                let result;

                try {
                    result = JSON.parse(text);
                } catch {
                    result = { message: text };
                }

                if (response.ok && result.success) {
                    msg.style.color = "green";
                    msg.textContent = "Login berhasil! Mengarahkan...";
                    setTimeout(() => {
                        window.location.href = "/home.html";
                    }, 1000);
                } else {
                    msg.textContent = result.message || "Username atau password salah.";
                }

            } catch (error) {
                msg.textContent = "Gagal terhubung ke server.";
                console.error("Login error:", error);
            }
        });
    </script>
